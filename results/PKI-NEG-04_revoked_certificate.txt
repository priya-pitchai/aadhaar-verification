from datetime import datetime, timedelta, timezone
from pathlib import Path

from cryptography import x509
from cryptography.x509.oid import NameOID
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import rsa

from verify_aadhaar import verify_certificate_revocation


TEST_DIR = Path("results/revoked_test")
TEST_DIR.mkdir(parents=True, exist_ok=True)


# -------------------------------------------------
# 1. Create synthetic CA key
# -------------------------------------------------

ca_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048
)

ca_name = x509.Name([
    x509.NameAttribute(
        NameOID.COUNTRY_NAME,
        "IN"
    ),
    x509.NameAttribute(
        NameOID.ORGANIZATION_NAME,
        "Research Test CA"
    ),
    x509.NameAttribute(
        NameOID.COMMON_NAME,
        "Synthetic Revocation Test CA"
    ),
])


# -------------------------------------------------
# 2. Create synthetic CA certificate
# -------------------------------------------------

now = datetime.now(timezone.utc)

ca_cert = (
    x509.CertificateBuilder()
    .subject_name(ca_name)
    .issuer_name(ca_name)
    .public_key(ca_key.public_key())
    .serial_number(x509.random_serial_number())
    .not_valid_before(now - timedelta(days=1))
    .not_valid_after(now + timedelta(days=365))
    .add_extension(
        x509.BasicConstraints(
            ca=True,
            path_length=None
        ),
        critical=True
    )
    .sign(
        ca_key,
        hashes.SHA256()
    )
)


# -------------------------------------------------
# 3. Create synthetic end certificate
# -------------------------------------------------

leaf_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048
)

leaf_name = x509.Name([
    x509.NameAttribute(
        NameOID.COUNTRY_NAME,
        "IN"
    ),
    x509.NameAttribute(
        NameOID.ORGANIZATION_NAME,
        "Research Test UIDAI"
    ),
    x509.NameAttribute(
        NameOID.COMMON_NAME,
        "Synthetic UIDAI Certificate"
    ),
])

leaf_serial = 123456789

leaf_cert = (
    x509.CertificateBuilder()
    .subject_name(leaf_name)
    .issuer_name(ca_name)
    .public_key(leaf_key.public_key())
    .serial_number(leaf_serial)
    .not_valid_before(now - timedelta(days=1))
    .not_valid_after(now + timedelta(days=30))
    .sign(
        ca_key,
        hashes.SHA256()
    )
)


# -------------------------------------------------
# 4. Create CRL containing leaf serial as REVOKED
# -------------------------------------------------

revoked = (
    x509.RevokedCertificateBuilder()
    .serial_number(leaf_serial)
    .revocation_date(
        now - timedelta(hours=1)
    )
    .build()
)

crl = (
    x509.CertificateRevocationListBuilder()
    .issuer_name(ca_name)
    .last_update(
        now - timedelta(hours=1)
    )
    .next_update(
        now + timedelta(days=7)
    )
    .add_revoked_certificate(
        revoked
    )
    .sign(
        private_key=ca_key,
        algorithm=hashes.SHA256()
    )
)


# -------------------------------------------------
# 5. Save test certificates and CRL
# -------------------------------------------------

ca_path = TEST_DIR / "test_ca.pem"
leaf_path = TEST_DIR / "test_leaf.pem"
crl_path = TEST_DIR / "test_revoked.crl"


ca_path.write_bytes(
    ca_cert.public_bytes(
        serialization.Encoding.PEM
    )
)

leaf_path.write_bytes(
    leaf_cert.public_bytes(
        serialization.Encoding.PEM
    )
)

crl_path.write_bytes(
    crl.public_bytes(
        serialization.Encoding.DER
    )
)


# -------------------------------------------------
# 6. Run our revocation function
# -------------------------------------------------

try:

    verify_certificate_revocation(
        uidai_cert_path=str(leaf_path),
        crl_path=str(crl_path),
        issuer_cert_path=str(ca_path),
    )

    print(
        "UNEXPECTED: Revoked certificate accepted"
    )


except Exception as exc:

    print("EXPECTED FAILURE")
    print(
        type(exc).__name__
        + ": "
        + str(exc)
    )
